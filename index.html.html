// 當前選擇的階段狀態

let currentTab = 'wonder';

let lastActivityIndex = -1;

// 切換頁籤函數

function switchTab(tabKey) {

currentTab = tabKey;

const data = albumsData[tabKey];

// 1. 更新頁籤按鈕樣式

document.querySelectorAll('.tab-btn').forEach(btn => {

btn.classList.remove('active', 'border-sky-400', 'border-pink-400', 'border-slate-600', 'border-purple-400', 'bg-sky-50/50', 'bg-pink-50/50', 'bg-slate-800/10', 'bg-purple-50/50');

});

const activeBtn = document.getElementById(btn-${tabKey});

activeBtn.classList.add('active');

// 2. 更新專輯卡片內容與背景

const albumCard = document.getElementById('album-card');

const cardContent = document.getElementById('card-content');

// 替換背景色

albumCard.className = rounded-3xl p-6 md:p-8 shadow-sm transition-all duration-500 mb-8 border border-white/60 ${data.bgClass};

// 動態卡片內容 HTML

cardContent.className = 'fade-in';

cardContent.innerHTML = `

<div class="flex flex-col md:flex-row justify-between items-start md:items-center mb-4 gap-2">

<div>

<span class="text-xs font-serif tracking-widest uppercase opacity-80 block">${data.englishTitle}</span>

<h2 class="text-2xl md:text-3xl font-bold">${data.title}</h2>

</div>

<span class="inline-block px-4 py-1.5 rounded-full text-xs font-bold tracking-wider ${data.tagColor} shadow-sm">

概念階段：${data.badge}

</span>

</div>

<p class="text-sm md:text-base leading-relaxed mb-6 opacity-90">

${data.concept}

</p>

<div class="grid md:grid-cols-2 gap-4 text-xs md:text-sm pt-4 border-t ${tabKey === 'tear' ? 'border-slate-700' : 'border-slate-200/60'}">

<div>

<span class="font-bold block mb-1">🎵 代表曲目 / 收錄曲：</span>

<p class="opacity-80">${data.tracks.join(' • ')}</p>

</div>

<div>

<span class="font-bold block mb-1">🎨 視覺與色彩風格：</span>

<p class="opacity-80">${data.colorStyle}</p>

</div>

</div>

`;

// 3. 更新推薦活動區塊標籤並更換新活動

const activityTag = document.getElementById('activity-stage-tag');

activityTag.className = text-xs px-3 py-1 rounded-full font-medium ${data.tagColor};

activityTag.innerText = ${data.badge} 階段;

// 重置索引並獲取新活動

lastActivityIndex = -1;

getRandomActivity();

}

// 隨機獲取推薦活動（避免連續重複）

function getRandomActivity() {

const activities = albumsData[currentTab].activities;

let newIndex;

if (activities.length > 1) {

do {

newIndex = Math.floor(Math.random() * activities.length);

} while (newIndex === lastActivityIndex);

} else {

newIndex = 0;

}

lastActivityIndex = newIndex;

const item = activities[newIndex];

const displayArea = document.getElementById('activity-display');

// 觸發淡入效果

displayArea.classList.remove('fade-in');

void displayArea.offsetWidth; // 觸發 reflow

displayArea.classList.add('fade-in');

displayArea.innerHTML = `

<div class="max-w-xl">

<h3 class="text-lg md:text-xl font-bold text-slate-800 mb-2 flex items-center justify-center gap-2">

<span>${item.title}</span>

</h3>

<div class="flower-line my-3 w-16 mx-auto"></div>

<p class="text-slate-600 text-sm md:text-base leading-relaxed italic">

「${item.reason}」

</p>

</div>

`;

}

// 頁面載入完成後初始化

window.onload = function() {

switchTab('wonder');
